<script>
    // =====================
    // ESTADO
    // =====================
    let selectedPlan = { name: 'basico', price: 49900 };
    let deferredPrompt;

    // =====================
    // TABS
    // =====================
    function switchTab(tab) {
        document.querySelectorAll('.tab').forEach((t, i) => {
            t.classList.toggle('active', (i === 0 && tab === 'login') || (i === 1 && tab === 'register'));
        });
        document.getElementById('loginForm').classList.toggle('active', tab === 'login');
        document.getElementById('registerForm').classList.toggle('active', tab === 'register');
        clearAlerts();
    }

    // =====================
    // LOGIN
    // =====================
    function handleLogin(e) {
        e.preventDefault();
        const email = document.getElementById('loginEmail').value;
        const pass = document.getElementById('loginPass').value;
        const remember = document.getElementById('rememberMe').checked;

        showLoading('loginLoading', 'loginBtn');

        setTimeout(() => {
            hideLoading('loginLoading', 'loginBtn', '🚀 Ingresar al Curso');

            // Verificar usuarios registrados
            const users = JSON.parse(localStorage.getItem('prp5g_users') || '[]');
            const user = users.find(u => u.email === email && u.password === pass);

            // Usuario demo
            if (email === 'demo@prp5g.com' && pass === 'demo123') {
                loginSuccess({ name: 'Usuario Demo', email, plan: 'premium' }, remember);
                return;
            }

            if (user) {
                // Verificar si pagó
                if (!user.paid) {
                    window.location.href = `payment.html?email=${encodeURIComponent(email)}&plan=${user.plan}&price=${user.planPrice}`;
                    return;
                }
                loginSuccess(user, remember);
            } else {
                showAlert('loginAlert', 'error', '❌ Correo o contraseña incorrectos');
            }
        }, 1500);
    }

    function loginSuccess(user, remember) {
        const storage = remember ? localStorage : sessionStorage;
        storage.setItem('prp5g_session', JSON.stringify(user));
        showAlert('loginAlert', 'success', `✅ ¡Bienvenida ${user.name}! Redirigiendo...`);
        setTimeout(() => { window.location.href = 'dashboard.html'; }, 1200);
    }

    // =====================
    // REGISTRO
    // =====================
    function handleRegister(e) {
        e.preventDefault();
        const name = document.getElementById('regName').value;
        const email = document.getElementById('regEmail').value;
        const phone = document.getElementById('regPhone').value;
        const pass = document.getElementById('regPass').value;

        if (pass.length < 6) {
            showAlert('registerAlert', 'error', '❌ La contraseña debe tener al menos 6 caracteres');
            return;
        }

        showLoading('registerLoading', 'registerBtn');

        setTimeout(() => {
            hideLoading('registerLoading', 'registerBtn', '✨ Crear Cuenta y Continuar al Pago');

            const users = JSON.parse(localStorage.getItem('prp5g_users') || '[]');

            if (users.find(u => u.email === email)) {
                showAlert('registerAlert', 'error', '❌ Este correo ya está registrado');
                return;
            }

            const newUser = {
                name, email, phone, password: pass,
                plan: selectedPlan.name,
                planPrice: selectedPlan.price,
                paid: false,
                createdAt: new Date().toISOString()
            };

            users.push(newUser);
            localStorage.setItem('prp5g_users', JSON.stringify(users));

            showAlert('registerAlert', 'success', '✅ ¡Cuenta creada! Redirigiendo al pago...');

            setTimeout(() => {
                window.location.href = `payment.html?email=${encodeURIComponent(email)}&name=${encodeURIComponent(name)}&plan=${selectedPlan.name}&price=${selectedPlan.price}`;
            }, 1200);
        }, 1500);
    }

    // =====================
    // PLAN SELECTOR
    // =====================
    function selectPlan(el, name, price) {
        document.querySelectorAll('.plan-card').forEach(c => c.classList.remove('selected'));
        el.classList.add('selected');
        selectedPlan = { name, price };
    }

    // =====================
    // SOCIAL LOGIN
    // =====================
    function socialLogin(provider) {
        showAlert('loginAlert', 'success', `✅ Conectando con ${provider}...`);
        setTimeout(() => {
            const fakeUser = { name: `Usuario ${provider}`, email: `user@${provider.toLowerCase()}.com`, plan: 'basico', paid: true };
            sessionStorage.setItem('prp5g_session', JSON.stringify(fakeUser));
            window.location.href = 'dashboard.html';
        }, 1500);
    }

    // =====================

    function openForgot() {
        document.getElementById('forgotModal').classList.add('active');
    }

    function sendRecovery() {
        const email = document.getElementById('forgotEmail').value;
        if (!email) return;
        const alert = document.getElementById('forgotAlert');
        alert.className = 'alert success';
        alert.innerHTML = '✅ Instrucciones enviadas a ' + email;
        alert.style.display = 'flex';
    }

    // =====================
    // HELPERS
    // =====================
    function togglePassword(inputId, btn) {
        const input = document.getElementById(inputId);
        input.type = input.type === 'password' ? 'text' : 'password';
        btn.textContent = input.type === 'password' ? '👁️' : '🙈';
    }

    function showAlert(id, type, msg) {
        const el = document.getElementById(id);
        el.className = 'alert ' + type;
        el.innerHTML = msg;
        el.style.display = 'flex';
    }

    function clearAlerts() {
        document.querySelectorAll('.alert').forEach(a => a.style.display = 'none');
    }

    function showLoading(loadId, btnId) {

        document.getElementById(loadId).style.display = 'none';
        const btn = document.getElementById(btnId);
        btn.style.display = 'block';
        btn.textContent = btnText;
